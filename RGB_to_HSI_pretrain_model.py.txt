import os
import numpy as np
import cv2
import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader
import matplotlib.pyplot as plt

# ============================================================
# 1️⃣ LOAD 31 BAND IMAGES → BUILD HSI CUBE
# ============================================================
def load_hsi_cube(folder, prefix):
    bands = []

    for i in range(1, 32):  # 01 to 31
        filename = f"{prefix}_{i:02d}.png"
        path = os.path.join(folder, filename)

        band = cv2.imread(path, cv2.IMREAD_GRAYSCALE)
        band = band.astype(np.float32) / 255.0
        bands.append(band)

    cube = np.stack(bands, axis=2)  # (H, W, 31)
    return cube


# ============================================================
# 2️⃣ DATASET CLASS
# ============================================================
class SSRDataset(Dataset):
    def __init__(self, root):
        self.root = root
        self.scenes = [d for d in os.listdir(root) if os.path.isdir(os.path.join(root, d))]


    def __len__(self):
        return len(self.scenes)

    def __getitem__(self, idx):
        scene = self.scenes[idx]

    # Example: beads_ms
        scene_folder = os.path.join(self.root, scene, scene)

        base = scene.replace("_ms", "")

    # RGB is inside SAME folder as bands
        rgb_path = os.path.join(scene_folder, f"{base}_RGB.bmp")

        if not os.path.exists(rgb_path):
            raise FileNotFoundError(f"RGB file not found: {rgb_path}")

        rgb = cv2.imread(rgb_path)
        rgb = cv2.cvtColor(rgb, cv2.COLOR_BGR2RGB) / 255.0

    # Load hyperspectral cube
        hsi = load_hsi_cube(scene_folder, scene)

        rgb = torch.tensor(rgb).permute(2,0,1).float()
        hsi = torch.tensor(hsi).permute(2,0,1).float()

        return rgb, hsi

# ============================================================
# 3️⃣ CNN MODEL (SSR NETWORK)
# ============================================================
class SSRNet(nn.Module):
    def __init__(self):
        super().__init__()

        self.conv1 = nn.Conv2d(3, 64, 3, padding=1)
        self.conv2 = nn.Conv2d(64, 128, 3, padding=1)
        self.conv3 = nn.Conv2d(128, 64, 3, padding=1)
        self.conv4 = nn.Conv2d(64, 31, 3, padding=1)

        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.relu(self.conv1(x))
        x = self.relu(self.conv2(x))
        x = self.relu(self.conv3(x))
        x = self.conv4(x)
        return x


# ============================================================
# 4️⃣ TRAINING
# ============================================================
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

dataset = SSRDataset(r"C:\Users\HARSHAA\Desktop\datasset_hyper")
loader = DataLoader(dataset, batch_size=1, shuffle=True)

model = SSRNet().to(device)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-4)
loss_fn = nn.MSELoss()

epochs = 50

print("Training on", len(dataset), "scenes")

for epoch in range(epochs):
    total_loss = 0

    for rgb, hsi in loader:
        rgb, hsi = rgb.to(device), hsi.to(device)

        pred = model(rgb)
        loss = loss_fn(pred, hsi)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        total_loss += loss.item()

    print(f"Epoch {epoch+1}/{epochs}, Loss: {total_loss:.4f}")

torch.save(model.state_dict(), "ssr_model.pth")
print("Model Saved!")

# ============================================================
# 5️⃣ TEST RGB → HYPERSPECTRAL
# ============================================================
model.eval()

test_scene = dataset.scenes[0]
scene_folder = os.path.join(dataset.root, test_scene, test_scene)
base = test_scene.replace("_ms", "")

test_rgb_path = os.path.join(scene_folder, f"{base}_RGB.bmp")

test_rgb = cv2.imread(r"C:\Users\HARSHAA\Desktop\dasa_foot.jpeg")
test_rgb = cv2.cvtColor(test_rgb, cv2.COLOR_BGR2RGB)/255.0

tensor = torch.tensor(test_rgb).permute(2,0,1).unsqueeze(0).float().to(device)

with torch.no_grad():
    output = model(tensor)

pred = output.squeeze(0).cpu().numpy().transpose(1,2,0)

np.save("predicted_hsi.npy", pred)
print("Predicted cube saved!")
